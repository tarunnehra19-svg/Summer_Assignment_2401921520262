// Question-2 Conatains duplicate

class Solution {
public:
    bool containsDuplicate(vector<int>& nums) {
        int i,end=nums.size()-1,mid,start=0;
        int x;
       sort(nums.begin(),nums.end());
       for(i=0;i<end;i++){
        if(nums[i]==nums[i+1]){
            return true;
        }
       }
       return false;
        
    }
};
