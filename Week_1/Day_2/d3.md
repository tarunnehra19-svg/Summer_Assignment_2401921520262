// Question-3 Maximum avg subarray


class Solution {
public:
    double findMaxAverage(vector<int>& nums, int k) {
        double max_avg,sum=0,avg=0;
        for(int i=0;i<k;i++){
            sum +=nums[i];
        }
        avg=sum/k;
        max_avg=avg;
        max_avg=max(max_avg,avg);
        for(int i=k;i<nums.size();i++){
            sum = sum - nums[i-k] + nums[i];
            avg=sum/k;
            max_avg=max(max_avg,avg);
        }
        return max_avg;
    }
};
