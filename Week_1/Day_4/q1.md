// Question-1 Sum of diagnol element in a matrix

class Solution {
public:
    int diagonalSum(vector<vector<int>>& mat) {
        int x=mat.size();
        int sum=0;
        for(int i=0;i<x;i++){
            for(int j=0;j<x;j++){
                if(i==j || i+j==x-1){
                 sum +=mat[i][j];
                }
            }
        }
        return sum;
    }
};
