// Question- Reverse words in a string|||

class Solution {
public:
    string reverseWords(string s) {
        string wrd="";
        string ans="";
        for(char ch:s){
            if(ch==' '){
                reverse(wrd.begin(),wrd.end());
                ans += wrd;
                wrd="";
                ans +=" ";
            }
            else{
                wrd +=ch;
            }
        }
        reverse(wrd.begin(),wrd.end());
        return ans + wrd;
    }
};
