Question-Valid anagram

class Solution {
public:
    bool isAnagram(string s, string t) {
        sort(s.begin(),s.end());
        sort(t.begin(),t.end());
        int x=s.size(),y=t.size();
        if(x==y){
        for(int i=0;i<s.size();i++){
            if(s[i]!=t[i]){
                return 0;
            }
        }
        }
        else{
            return 0;
        }
        return 1;
    }
};
