Question-First Unique character in string

class Solution {
public:
    int firstUniqChar(string s) {
        vector<int>r(26,0);
        int i=0;
        for(int x=0;x<s.size();x++){
            r[s[x]-'a']++;
        }
        for(int x=0;x<s.size();x++){
            if(r[s[x]-'a']==1){
                i=x;
                return i;
            }
        }
     return -1;   
    }
};
